canvas.width = window.innerWidth;
canvas.height = window.innerHeight;

let particles = [];
const numParticles = 3000;
const pink = '#ff69b4';
let animationPhase = 'heart';
let phaseStartTime = Date.now();

class Particle {
    constructor(x, y) {
        this.x = x;
        this.y = y;
        this.size = Math.random() * 2 + 0.5;
        this.color = pink;
        this.speed = Math.random() * 0.05 + 0.01;
        this.targetX = x;
        this.targetY = y;
        this.originalX = x;
        this.originalY = y;
    }

    draw() {
        ctx.fillStyle = this.color;
        ctx.beginPath();
        ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
        ctx.closePath();
        ctx.fill();
    }

    update() {
        let dx = this.targetX + canvas.width / 2 - this.x;
        let dy = this.targetY + canvas.height / 2 - this.y;
        this.x += dx * this.speed;
        this.y += dy * this.speed;
    }
}

function createHeart() {
    particles = [];
    for (let i = 0; i < numParticles; i++) {
        let t = Math.random() * (2 * Math.PI);

        let y = 13 * Math.cos(t) - 5 * Math.cos(2 * t) - 2 * Math.cos(3 * t) - Math.cos(4 * t);
        let scale = 10;

        particles.push(new Particle(offsetX, offsetY));
    }
}

    ctx.textAlign = 'center';
    ctx.fillStyle = pink;
    ctx.clearRect(0, 0, canvas.width, canvas.height);
    ctx.fillText(text, canvas.width / 2, canvas.height / 2);
    
    const data = ctx.getImageData(0, 0, canvas.width, canvas.height).data;
    ctx.clearRect(0, 0, canvas.width, canvas.height);

    let textParticles = [];
    for (let i = 0; i < canvas.width; i += 5) {
        for (let j = 0; j < canvas.height; j += 5) {

    // Distribute existing particles to new text targets
    particles.forEach((p, index) => {
        const target = textParticles[index % textParticles.length];
        p.targetX = target.x;
        p.targetY = target.y;
        p.speed = Math.random() * 0.05 + 0.01;
    });
}
